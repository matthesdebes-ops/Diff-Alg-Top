\section{Covering Spaces}
\begin{definition}\label{3.1}
    Let $p:E\to B$ be a continuous, surjective map and let $U\subseteq B$ be an open set. We say $U$ is \textbf{evenly covered by p} if $p^{-1}(U)$ is the disjoint union of open sets $V_\alpha$ in $E$ such that $p|_{V_\alpha}$ is a homeomorphism from $V_\alpha$ onto $U$ for all $\alpha$.\\
    We say $\{V_\alpha\}_{\alpha\in A}$ is a partition of $p^{-1}$ into slices.
\end{definition}
\begin{definition}\label{3.2}
    Let $p:E\to B$ be a continuous, surjective map. If every $b\in B$ has a neighbourhood $U$ that is evenly covered by $p$, then $p$ is called a \textbf{covering map}\index{covering map}, $B$ is called the \textbf{base of the covering} and $E$ is the \textbf{covering Space of $B$}.\\
    We call $p^{-1}(x)$ a fiber over $x$.
\end{definition}
\begin{remark}\label{3.3}
    \leavevmode
    \begin{itemize}
        \item $\forall b\in B,p^{-1}(b)$ is discrete in $E$: each slice $V_\alpha$ is open in $E$ and intersects $p^{-1}(b)$ in a single point.
        \item Covering maps are open: Let $V\subseteq E$ be open. Take $x\in p(V)$. Choose a neighbourhood $U$ of $x$ in $B$ which is evenly covered by $p$. Let $\{V_\alpha\}$ be the slices of $p^{-1}(U)$ and choose $y\in V$ such that $p(y)=x$ and $\beta$ such that $y\in V_\beta$. We know that $V\cap V_\beta$ is open in $E$, hence open in $V_\beta$. Now we know that $p|_{V_\beta}:V_\beta\to U$ is a homeomorphism, hence $p(V\cap V_\beta)$ is open in $U$, hence open in $B$ and it is still a neighbourhood of $x$ in $p(V)$.
    \end{itemize}
\end{remark}
\begin{example}\label{3.4}
    \leavevmode
    \begin{itemize}
        \item Let $X$ be a topological space and $E=X\times \{1,...,n\}$. Then $p:E\to X$ given by $p(x,i)=x$ is a covering map.
        \item $p:\mathbb{R}\to \mathbb{S}^1$ given by $p(x)=(\cos(2\pi x),\sin(2\pi x))$ is a covering map. For example $p^{-1}(U)=\{x\in \mathbb{R}:\cos (2\pi x)>0\}=\bigcup\limits_{n\in \mathbb{Z}}(n-\frac{1}{4},n+\frac{1}{4})$, so we see that $V_n=(n-\frac{1}{4},n+\frac{1}{4})$ are disjoint open sets. Because $\sin(2\pi x)$ is strictly monoton on $\bar{V_n}$ we get that $p|_{\bar V_n}:\bar V_n\to \bar U$ is bijective and also homeomorphic.
        \item Conversely $p:\mathbb{R}^+\to \mathbb{S}^1$ with the same definition is \textbf{NOT} a covering map (despite being a surjective, continuous, local homeomorphism). This is due to $(1,0)$ not having a neighbourhood which is evenly covered by $p$ so $p:V_n\to U$ is a homeomorphism for all $n\neq 0$.
    \end{itemize}
\end{example}
\begin{theorem}\label{3.5}
    Let $p:E\to B$ and $p':E'\to B'$ be two covering maps. Then $p\times p':E\times E'\to B\times B'$ is again a covering map.
\end{theorem}
\begin{proof}
    Let $b\in B,b'\in B'$ and choose $U,U'$ to be the, by $p$ and $p'$ evenly covered neighbourhoods of $b$ and $b'$ respectively. So we get $p^{-1}(U)=\bigcup\limits_\alpha V_\alpha$ and $p'^{-1}(U')=\bigcup\limits_\beta V'_\beta$. 
    It follows that $(p\times p')^{-1}(U\times U')=(\bigcup\limits_{\alpha,\beta}V_\alpha\times V'_\beta)$ which shows that $p\times p'$ is indeed a covering map. 
\end{proof}
\begin{example}\label{3.6}
    \begin{itemize}
        \item $T=\mathbb{S}^1\times \mathbb{S}^1$ is the torus which has by the preciding theorem \ref{3.5} a covering map in $p\times p:\mathbb{R}\times \mathbb{R}\to \mathbb{S}^1\times \mathbb{S}^1$ given by $(s,t)\to (\cos(2\pi s),\sin(2\pi s),\cos(2\pi t),\sin(2\pi t))$.
        \item Let $p\times p$ be as above and let $b_0=p(0)=(1,0)$. We also define $B_0:=(\mathbb{S}^1\times\{b_0\})\cup(\{b_0\}\times \mathbb{S}^1)$ and $E_0:=(p\times p)^{-1}(B_0)=(\mathbb{R}\times\mathbb{Z})\cup(\mathbb{Z}\times \mathbb{R})$. Then $p\times p:E_0\to B_0$ is a covering map.
    \end{itemize}
\end{example}
\begin{definition}\label{3.7}
    Let $p:E\to B$ and $f:X\to B$ be a continuous map with $E,B,X$ being topological spaces and $p$ being a covering map.
    A \textbf{lifting}\index{lifting} of $f$ is a map $\tilde f:X\to E$ s.t. $p\circ \tilde f=f$
\end{definition}
\begin{example}\label{3.8}
    We choose $p:\mathbb{R}\to \mathbb{S}^1$ as before and $f:[0,1]\to \mathbb{S}^1$ given by $f(t)=(\cos(\pi t),\sin(\pi t))$. Then $f$ lifts to $\tilde f:[0,1]\to \mathbb{R}$, $t\to \frac{1}{2}t$.
    If we look at $h:[0,1]\to \mathbb{S}^1$ given by $h(t)=(\cos(4\pi t),\sin(4\pi t))$ we see that it lifts to $\tilde h(s)=2s$.
\end{example}
\begin{lemma}\label{3.9}
    Let $p:E\to B$ be again a covering map, $b_0\in B$ and $p(e_0)=b_0$.
    Any path $f:[0,1]\to B$ beginning at $b_0$ has a unique lifting to a path $\tilde f$ beginning at $e_0$.
\end{lemma}
\begin{proof}
    Let $B$ be covered by finitley many open sets $U$ which in turn are evenly covered by the map $p$ (possible because $B=f([0,1])$, $f$ is continuous, $[0,1]$ is compact and covered by the open sets $f^{-1}(U)$). Now we may choose a subdivision $0=s_0<s_1<...<s_n=1$ of $[0,1]$ such that $f([s_i,s_{i+1}])\subseteq U$.\\
    Set $\tilde f(0)=e_0$. Suppose $\tilde f$ is defined for $0\leq s\leq s_i$. We define $\tilde f$ on $[s_i,s_{i+1}]$ as follows:\\
    By construction $f([s_i,s_{i+1}])$ lies in some open $U$ which is evenly covered by $p$ with $p^{-1}(U)=\bigcup\limits_\alpha V_\alpha$ and $p:V_\alpha\to U$ is homeomorphic for all $\alpha$. We choose $\beta$ such that $\tilde f(s_i)\in V_\beta$ and set $\tilde f(s)=(p|_{V_\beta}^{-1}\circ f(s))$. Now it follows that $\tilde f$ is continuous and $p\circ \tilde f=f$ by definition.\\
    Uniqueness: Let $\tilde f'$ be another lifting with $\tilde f'(0)=e_0$. Suppose $\tilde f(s)=\tilde f'(s)\quad \forall 0\leq s\leq s_i$ and define $V_\beta$ as before.
    $\tilde f'([s_i,s_{i+1}])$ is connected, hence it must lie entirely in $V_\beta$. Thus for $s\in[s_i,s_{i+1}]$, $\tilde f'(s)$ equals to some point $y\in V_\beta$ such that $p(\tilde f'(s))=f(s)$ with $y\in p^{-1}(f(s))$. But there is only one such point, namely $\tilde f(s)\implies \tilde f'(s)=\tilde f(s)$.
\end{proof}
\begin{lemma}\label{3.10}
    Let $p:E\to B$ be a covering map and $p(e_0)=b_0$. Let $F:[0,1]\times[0,1]\to B$ be a continuous map with $F(0,0)=b_0$.
    Then there exists a unique lifting $\tilde F:[0,1]\times [0,1]\to E$ such that $\tilde F(0,0)=e_0$. If $F$ is a path homotopy, then so is $\tilde F$.
\end{lemma}
\begin{proof}
    We only provied a sketch here because this proof is very similar to the proof of the preceeding lemma \ref{3.9}. By set lemma we get that $F$ lifts to $I\times \{0\}$ and $\{0\}\times I$.
    Now we impose two different subdivisions $s_n$ and $t_m$ as before. If $F$ is a path homotopy, then $F(\{0\}\times I)=b_0$ and because $\tilde F$ is a lifting, it follows that $\tilde F(\{0\}\times I)\subseteq p^{-1}(b_0)$ which is discrete as we discussed.
    Because $\{0\}\times I$ is connected, it follows that $\tilde F(\{0\}\times I)$ is connected, hence $\tilde F(\{0\}\times I)$ is a one-point set. The same holds for $\tilde F(I\times \{0\})$.
\end{proof}
\begin{theorem}\label{3.11}
    Let $p:E\to B$ be a covering map with $p(e_0)=b_0$.
    Let $f,g$ be two paths in $B$ from $b_0$ to $b_1$ and let $\tilde f,\tilde g$ be the liftings of $f,g$ starting at $e_0$.
    If $f\simeq g$, then $\tilde f$ and $\tilde g$ end and start at the same point, and $\tilde f\simeq\tilde g$.
\end{theorem}
\begin{proof}
    Let $F:I\times I\to B$ be the path homotopy between $f,g$ so $F(0,0)=b_0$. By lemma \ref{3.10} $\tilde F$ is a lifting of $F$ to $E$ with $\tilde F(0,0)=e_0$ and it is also a path homotopy.
    We know therefore that $\tilde F(\{0\}\times I)=\{e_0\}$ and $\tilde F(\{0\}\times I)=\{e_1\}$.
    $\tilde F|_{I\times \{0\}}$ is a path starting at $e_0$ and the lifting of $F|_{I\times \{0\}}$. By the uniqueness of liftings $\tilde F(s,0)=\tilde f(s)$.
    The same argument gives $\tilde F(s,1)=\tilde g(s)$ which implies that $\tilde f(1)=\tilde g(1)=e_1$. Therefore $\tilde f\simeq_{\tilde F}\tilde g$.
\end{proof}
\begin{definition}\label{3.12}
    Let $p:E\to B$ be a covering map with $p(e_0)=b_0$.
    Choose $[f]\in \pi_1(B,b_0)$. Let $\tilde f$ be the lifting of $f$ such that $\tilde f(0)=e_0$.
    $\phi:\pi_1(B,b_0)\to p^{-1}(b_0)=E_{b_0}$ defined by $\phi([f]):=\tilde f(1)$. This $\phi$ is well defined due to the theorem \ref{3.11}.
\end{definition}
\begin{theorem}\label{3.13}
    In the same setup as in the definition \ref{3.12}:
    If $E$ is path-connected then $\phi$ is surjective.\\
    If $E$ is simply connected then $\phi$ is bijective.
\end{theorem}
\begin{proof}
    First, let $E$ be path-connected and $e_1\in p^{-1}(b_0)$. $\tilde f$ is a path in $E$ from $e_0$ to $e_1$. $f:=p\circ \tilde f$ is a loop in $B$ based at $b_0$. Then $\phi([f])=\tilde f(1)=e_1\implies \phi$ is surjective.\\
    Now let $E$ be simply connected and let $[f],[g]$ such that $\phi([f])=\phi([g])$. Then $\tilde f,\tilde g$ are liftings that start in $e_0$. It follows that $\tilde f(1)=\tilde g(1)$, hence $\tilde f\simeq_{\tilde F}\tilde g\implies f\simeq_{p\circ\tilde F}g$ 
\end{proof}

\subsection{Fundamental group of circles and applications}
\begin{theorem}\label{3.14}
    $\pi_1(\mathbb{S}^1)\simeq\mathbb{Z}$
\end{theorem}
\begin{proof}
    Let $b_0=(1,0),e_0=0$ and $p:\mathbb{R}\to\mathbb{S}^1$ be the standard covering.\\
    We need to show that $\phi:\pi_1(\mathbb{S}^1,b_0)\to p^{-1}(b_0)=\mathbb{Z}$ (from definition \ref{3.12}) is a homomorphism.\\
    For that, let $[f],[g]\in\pi_1(\mathbb{S}^1,b_0)$ and let $\tilde f,\tilde g$ be liftings starting at $0$ and define $n:=\tilde f(1),m:=\tilde g(1)$ ($n\in \mathbb{Z}$ because $p(\tilde f(1))=f(1)=b_0$ so $\tilde{f}(1)\in p^{-1}(b_0)=\mathbb{Z}$).
    Now we define $\tilde g'(s):= n+\tilde g(s)$ which is a path in $\mathbb{R}$. We have that $\forall x\in\mathbb{R},n\in\mathbb{Z}: p(x+n)=p(x)\implies \tilde g'(s)$ is the lifting of $g$ starting at $n$.
    It follows that $\tilde f*\tilde g'$ is well defined and a lifting of $f*g$ starting at $0$. Since $\tilde f*\tilde g'(1)=n+m$, we have $\phi([f]*[g])=n+m=\phi([f])+\phi([g])$.
\end{proof}
\begin{definition}\label{3.15}
    Let $A\subseteq X$. A \textbf{retraction}\index{retraction} of $X$ onto $A$ is a continuous map $r:X\to A$ such that $r|_A=id_A$. $A$ is then called \textbf{retract} of $X$.
\end{definition}
\begin{example}\label{3.16}
    $r:\mathbb{R}^2\setminus\{(0,0)\}\to \mathbb{S}^1$ with $r(x)=\frac{x}{\Vert x\Vert}$
\end{example}
\begin{definition}\label{3.17}
    For any continuous map $\varphi:(X,x_0)\to (Y,y_0)$ the \textbf{induced homomorphism} (or induced homomorphism of fundamental groups) is $\varphi_*:\pi_1(X,x_0)\to\pi_1(Y,y_0)$ given by $\varphi_*([f])=[\varphi\circ f]$.
\end{definition}
\begin{lemma}\label{3.18}
    If $r:X\to A$ is a retraction, then the homomorphism of fundamental groups induced by the inclusion $j:A\to X$ is injective.
\end{lemma}
\begin{proof}
    We know $r\circ j=id_A\implies r_*\circ j_*=(r\circ j)_*=(id_A)_*=id$ with $j_*([f])=[j\circ f]$.\\
    Now, because $j_*$ has a left inverse we get that it is also injective.
\end{proof}
\begin{theorem}\label{3.19}
    There is no retraction $r:D^2\to\mathbb{S}^1$ with $D^2=\{x\in \mathbb{R}^2|\Vert x\Vert\leq 1\}$.
\end{theorem}
\begin{proof}
    Assume $\mathbb{S}^1$ is a retract of $D^2$ and let $j:\mathbb{S}^1\to D^2$ be the inclusen.\\
    Then $j_*:\pi_1(\mathbb{S}^1,x_0)\to\pi_1(D^2,x_0)$ is injective.\\
    But this is a contradiction because $\pi_1(\mathbb{S}^1,x_0)=\mathbb{Z}$ and $\pi_1(D^2,x_0)=0$.
\end{proof}
\begin{lemma}\label{3.20}
    Let $h:\mathbb{S}^1\to X$ be continuous. The following are equivalent:
    \begin{enumerate}
        \item $h$ is nullhomotopic
        \item $h$ extends to a continuous map $\tilde h:D^2\to X$
        \item $h_*=0$ ($h$ induces the trivial homomorphism on the fundamental groups)
    \end{enumerate}
\end{lemma}
\begin{proof}
    $(i)\implies (ii)$\\
    $(i)\implies\exists H:\mathbb{S}^1\times I\to X$ homotopy between $h$ and a constant path.\\
    Now define $\pi:\mathbb{S}^1\times I\to D^2$ as $\pi(x,t)=(1-t)x$. This function is continuous, closed and surjective.\\
    $\pi$ collapses $\mathbb{S}^1\times \{1\}$ to $(0,0)$ and is injective everywhere else.\\
    We know that $H$ is constant on $\mathbb{S}^1\times 1\implies H\circ \pi^{-1}$ is well-defined, so this gives a continuous extension $\tilde h$ of $h$ to $D^2$.\\
    $(ii)\implies (iii)$\\
    Let $\tilde h$ be the continuous extension of $h$ given by $(ii)$ and let $j:\mathbb{S}^1\to D^2$ be the inclusion map.\\
    $h=\tilde h\circ j\implies h_*=\tilde h_*\circ j_*$ and $j_*:\pi_1(\mathbb{S}^1,x_0)\to\pi_1(D^2,x_0)=0\implies j_*=0\implies h_*=0$.\\
    $(iii)\implies (i)$\\
    $\pi_1(\mathbb{S}^1,b_0)$ is generated by $[p|_I]$, $p_0=p|_I$ and we define $x_0:=h(b_0)$.\\
    $(iii)$ gives $0=h_*[p_0]=[h\circ p_0]$, hence there is a homotopy $F$ in $X$ between $h\circ p_0$ and $e_{x_0}$ with $F(\{0\}\times I)=\{x_0\},F(\{1\}\times I)=\{x_0\}$.\\
    It follows that $(s,t)\to F(p_0^{-1}(s),t)$ is well-defined on $\mathbb{S}^1\times I$ and defines a homotopy between $h$ and $e_{x_0}$.
\end{proof}
\begin{corollary}\label{3.21}
    The inclusion $j:\mathbb{S}^1\to\mathbb{R}^2\setminus\{(0,0)\}$ is not nullhomotopic and the identity $i:\mathbb{S}^1\to\mathbb{S}^1$ is not nullhomotopic too.
\end{corollary}
\begin{proof}
    \begin{itemize}
        \item Let $r:\mathbb{R}^2\setminus\{(0,0)\}\to\mathbb{S}^1$ given by $x\to \frac{x}{\Vert x\Vert}$ be a retraction.\\
        \item It follows that $j_*$ is injective hence non trivial so by lemma \ref{3.20} $j$ is not nullhomotopic.
        \item Let $i$ be the identity on $\mathbb{S}^1\implies I_*\neq 0\implies i$ is not nullhomotopic.
    \end{itemize}
\end{proof}
\begin{definition}\label{3.22}
    A \textbf{vectorfield}\index{vectorfield} on $D^2$ is a pair $(x,V(x))$ with $x\in D^2$ and $V(x)\in \mathbb{R}^2$ where $V:D^2\to\mathbb{R}^2$ is continuous.\\
    $V$ points directly inward at $x$ if $V(x)=\alpha x$ with $\alpha<0$ and directly outward if $\alpha>0$. 
\end{definition}
\begin{theorem}\label{3.23}
    Given a \textbf{nonvanishing} vectorfield $V$ on $D^2$, there exists a point on $\mathbb{S}^1$ where $V$ points directly outward and one where $V$ points directly inward.
\end{theorem}
\begin{proof}
    $V:D^2\to\mathbb{R}^2\setminus\{(0,0)\}$ is continuous.\\
    Assume that $V(x)$ does not point directly inward $\forall x\in \mathbb{S}^1$.\\
    We define $w:=V|_{\mathbb{S}^1}:\mathbb{S}^1\to\mathbb{R}^2\setminus\{(0,0)\}$ which is nullhomotopic (Lemma \ref{3.20} $(ii)$).\\
    But $w$ is homotopic to the inclusion $j:\mathbb{S}^1\to\mathbb{R}^2\setminus\{(0,0)\}$ which we show by the following means.\\
    So lets define $F(x,t)=tx+(1-t)w(x)$ which is continuous for $x\in \mathbb{S}^1$ and we have \\
    $F(x,0)=w(x)\neq 0$, $F(x,1)=x\neq 0$
    If $F(x,t)=0$ for some $t\in(0,1)$,\\
    then $w(x)=\frac{-tx}{1-t}=-\frac{t}{1-t}x$ points directly inward. \\
    Hence $F(x,t)\neq 0$ for $x\in\mathbb{S}^1,t\in I\implies F$ 
    is a homotopy between $w$ and $j$ in $\mathbb{R}^2\setminus\{(0,0)\}\implies j$ is nullhomotopic $\lightning$\\
    If we replace $V$ by $-V$ we get a point in $\mathbb{S}^1$, where $V$ points directly outward.
\end{proof}
\begin{theorem}[Brenwer fixed point theorem]\label{3.24}
    If $f:D^2\to D^2$ is continuous, then it has at least one fixed point.
\end{theorem}
\begin{proof}
    By contradiction, assume $\forall x\in D^2:f(x)\neq x$. Then $V(x):=f(x)-x$ is a non vanishing vectorfield on $D^2$.
    $V$ cannot point directly outward at some $x\in \mathbb{S}^1$:\\
    $V(x)=\alpha x$ with $\alpha>0\implies f(x)=x+\alpha x=(1+\alpha)x\notin D^2$$\lightning$
\end{proof}

    \[j_*:\pi_1(V,x_0)\to\pi_1(X,x_0)\]

\begin{proof}
    $1$st step: Show that there exists a subdivision $a_0<...<a_n$ of $I$ such that $\forall i:f(a_i)\in U\cap V$ and $f([a_i,a_{i+1}])$ lies in $U$ or $V$.\\
    Start with a subdivision $b_0<...<b_m$ of $I$ such that $f([b_i,b_{i+1}])$ lies in $U$ or $V$ \\
    for all $i$.
    \begin{itemize}
        \item If $f(b_i)\in U\cap V$ then we are done.
        \item If $f(b_i)\notin U\cap V$: $f([b_{i-1},b_i])$ and $f([b_i,b_{i+1}])$ lie entierly in $U$ or $V$ but because $f(b_i)\notin U\cap V\implies f([b_{i-1},b_{i+1}])$ lies in $U$ or $V$ so just remove $b_i$ from the subdivision.
    \end{itemize}
    $2$nd step: Let $l_i$ be a positive linear map from $[0,1]$ to $[a_{i-1},a_i]$ and define $f_i:=f\circ l_i$ which is a path either in $U$ or $V$.\\
    We have $[f]=[f_1]*...*[f_n]$ for all $i$. Now choose a path $\gamma_i$ in $U\cap V$ connecting $x_0$ and $f(a_i)$ so $\gamma_0=\gamma_n=e_{x_0}$.\\
    Then $g_i:=\gamma_{i-1}*f_i*\bar{\gamma_i}$ is a loop at $x_0$ lying in $U$ or $V$ and $[g_1]*...*[g_n]=[f_1]*...*[f_n]=[f]$.\\
\end{proof}
\begin{corollary}\label{3.26}
    Let $X=U\cup V$ with $U,V$ open and $U\cap V$ path-connected. If $U,V$ are simply connected then $X$ is too.
\end{corollary}
\begin{proof}
    $\pi_1(U,x_0)=0=\pi_1(V,x_0)\stackrel{\ref{3.25}}{\implies} \pi_1(X,x_0)$
\end{proof}
\begin{theorem}\label{3.27}
    For $n\geq 2$, $\mathbb{S}^n$ is simply connected.
\end{theorem}
\begin{proof}
    Define the following: $\mathbb{S}^n=U\cup V$ where $U=\mathbb{S}^n\setminus\{p\}$, $V=\mathbb{S}^n\setminus\{q\}$ ($p=(0,...,0,1)$, $q=(0,...,0,-1)$) and so $U\cap V=\mathbb{S}^n\setminus\{p,q\}$.\\
    $U,V$ are simply connected because $U\simeq\mathbb{R}^n$ and $V\simeq\mathbb{R}^n$ which follows from the stereographic projection:\\
    $f:\mathbb{S}^n\setminus\{p\}\to\mathbb{R}^n$ with $f(x)=\frac{1}{1-x_{n+1}}(x_1,...,x_n)$ is a homeomorphism and\\
    $g:\mathbb{R}^n\to\mathbb{S}^n\setminus\{p\}$ defined by $g(y)=(t(y)y_1,...,t(y)y_n,1-t(y))$ with $t(y)\frac{2}{1+\Vert y\Vert^2}$ is its inverse.\\
    We see that $U\cap V=\mathbb{S}^n\setminus\{p,q\}\simeq\mathbb{R}^n\{0\}$ is path connected if $n\geq 2$.\\
    Then with corollary \ref{3.26} it follows that $\mathbb{S}^n$ is simply connected.
\end{proof}
\begin{theorem}\label{3.28}
    $\pi_1(X\times Y,x_0\times y_0)$ is isomorphic to $\pi_1(X,x_0)\times\pi_1(Y,y_0)$.
\end{theorem}
\begin{proof}
    Let $p:X\times Y\to X$ and $q:X\times Y\to Y$ be the natural projections.
    \[p_*:\pi_1(X\times Y,x_0\times y_0)\to\pi_1(X,x_0)\]
    \[q_*:\pi_1(X\times Y,x_0\times y_0)\to\pi_1(Y,y_0)\]
    Define $\phi:\pi_1(X\times Y,x_0\times y_0)\to\pi_1(X,x_0)\times\pi_1(Y,y_0)$ by $\phi([f])=p_*([f])\times q_*([f])$.
    We now show that $\phi$ is an isomorphism.
    \begin{itemize}
        \item $\phi$ is surjective: Let $g$ be a loop in $X$ based at $x_0$ and $h$ be a loop in $Y$ based at $y_0$.\\
        $f:I\to X\times Y$ defined by $f(t)=g(t)\times h(t)$ is then a loop in $X\times Y$ based at $x_0\times y_0$ with\\
        $\phi([f(t)])=[p\circ f]\times[q\circ f]=[g]\times [h]$.
        \item $\phi$ is injective: Let $f$ be a loop in $X\times Y$ based at $x_0\times y_0$ with $\phi([f])=[e_{x_0}]\times[e_{y_0}]$.\\
        Hence $p\circ f\simeq_Ge_{x_0}$ and $q\circ f\simeq_He_{y_0}$ so $F:I\times I\to X\times Y$, $F(s,t):=G(s,t)\times H(s,t)$ defines a homotopy between $f$ and $e_{x_0\times y_0}$ so $[f]=0$.
    \end{itemize}
\end{proof}
\begin{corollary}\label{3.29}
    $\pi_1(\mathbb{S}^1\times\mathbb{S}^1,x_0\times y_0)$ is isomorphic to $\mathbb{Z}\times\mathbb{Z}$.\\
    In particular, $\mathbb{S}^2$ and $\mathbb{T}^2=\mathbb{S}^1\times\mathbb{S}^1$ are not homeomorphic.
\end{corollary}
\begin{theorem}\label{3.30}
    $\mathbb{R}^2$ is not homeomorhic to $\mathbb{R}^3$
\end{theorem}
\begin{proof}[1]
    Assume they are, then $\mathbb{R}^2\setminus\{0\}$ and $\mathbb{R}^3\setminus\{0\}$ are too.
    \begin{itemize}
        \item $\mathbb{R}^2\setminus\{0\}$ is not simpy connected because $j:\mathbb{S}^1\to\mathbb{R}^2\setminus\{0\}$ is not nullhomotopic.
        \item $\mathbb{R}^3\setminus\{0\}$ is simply connected: \\
        Fix $x_0\in\mathbb{S}^2$ and $[f]\in\pi_1(\mathbb{R}^3\setminus\{0\},x_0)$.\\
        Define $H(s,t)=(1-t)f(s)+t\frac{f(s)}{\Vert f(s)\Vert}=((1-t)+\frac{t}{\Vert f\Vert})f(s)\neq 0$.\\
        Hence $H$ is a homotopy in $\mathbb{R}^3\setminus\{0\}$ between $f$ and $f'=\frac{f}{\Vert f\Vert}$ which is a loop in $\mathbb{S}^2$.\\
        Because $\pi_1(\mathbb{S}^2,x_0)=0\implies f'\simeq e_{x_0}\implies [f]=0$.    
    \end{itemize}
\end{proof}
\begin{proof}[2]
    Assume they are, then $\mathbb{R}^2\setminus\{0\}$ and $\mathbb{R}^3\setminus\{0\}$ are too.
    We define $\phi:\mathbb{R}^n\setminus\{0\}\to\mathbb{S}^{n-1}\times\mathbb{R}$ by $\phi(x)=(\frac{x}{\Vert x\Vert},ln(\Vert x\Vert))$.\\
